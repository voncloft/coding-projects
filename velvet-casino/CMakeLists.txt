cmake_minimum_required(VERSION 3.20)
project(VelvetCasino VERSION 1.0.17 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network Multimedia)
add_library(casino_core src/game.cpp)
target_include_directories(casino_core PUBLIC src)
target_link_libraries(casino_core PUBLIC Qt6::Core)
# Default builds contain freely redistributable CC0 recordings. A local optional
# sound pack is embedded into the game, never downloaded at build/run time.
set(CASINO_SOUND_DIR "" CACHE PATH "Optional folder containing six custom WAV effects and intro.wav")
set(sound_qrc "<RCC><qresource prefix=\"/sounds\">\n")
foreach(sound chip-bet chip-payout chip-allin card-deal card-shuffle deck-tap intro)
  set(sound_file "${CMAKE_CURRENT_SOURCE_DIR}/assets/${sound}.wav")
  if(CASINO_SOUND_DIR)
    set(sound_file "${CASINO_SOUND_DIR}/${sound}.wav")
  endif()
  if(NOT EXISTS "${sound_file}")
    message(FATAL_ERROR "Sound pack is missing ${sound_file}")
  endif()
  string(APPEND sound_qrc "<file alias=\"${sound}.wav\">${sound_file}</file>\n")
endforeach()
string(APPEND sound_qrc "</qresource></RCC>\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sounds.qrc" "${sound_qrc}")
add_library(casino_ui src/window.cpp src/table.cpp src/network.cpp src/sounds.cpp "${CMAKE_CURRENT_BINARY_DIR}/sounds.qrc")
target_link_libraries(casino_ui PUBLIC casino_core Qt6::Widgets Qt6::Network Qt6::Multimedia)
target_compile_definitions(casino_ui PUBLIC CASINO_VERSION="${PROJECT_VERSION}")
foreach(target casino_core casino_ui)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
add_executable(velvet-casino src/main.cpp assets/resources.qrc)
target_link_libraries(velvet-casino PRIVATE casino_ui)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(casino_tests tests/tests.cpp)
  target_link_libraries(casino_tests PRIVATE casino_ui Qt6::Test)
  add_test(NAME casino_rules_network_ui COMMAND casino_tests)
  set_tests_properties(casino_rules_network_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS velvet-casino RUNTIME DESTINATION bin)
install(FILES assets/velvet-casino.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES LICENSE README.md THIRD_PARTY_NOTICES.txt DESTINATION share/doc/velvet-casino)
install(FILES licenses/Kenney-Casino-Audio.txt licenses/Freesound-Intro.txt licenses/Pixabay-Sounds.md DESTINATION share/doc/velvet-casino/licenses)
