cmake_minimum_required(VERSION 3.20)
project(TurnoverGolf VERSION 1.0.8 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
add_library(golf_engine src/game.cpp)
target_include_directories(golf_engine PUBLIC src)
target_link_libraries(golf_engine PUBLIC Qt6::Core)
target_compile_options(golf_engine PRIVATE -Wall -Wextra -Wpedantic)
add_library(golf_ui src/window.cpp src/cards.cpp src/network.cpp assets/resources.qrc)
target_link_libraries(golf_ui PUBLIC golf_engine Qt6::Widgets Qt6::Network)
target_compile_definitions(golf_ui PUBLIC GOLF_VERSION="${PROJECT_VERSION}")
target_compile_options(golf_ui PRIVATE -Wall -Wextra -Wpedantic)
add_executable(turnover-golf src/main.cpp)
target_link_libraries(turnover-golf PRIVATE golf_ui)
if(BUILD_TESTING)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
  add_executable(golf_tests tests/game_test.cpp)
  target_link_libraries(golf_tests PRIVATE golf_ui Qt6::Test)
  add_test(NAME golf_rules_and_ui COMMAND golf_tests)
  set_tests_properties(golf_rules_and_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(network_tests tests/network_test.cpp)
  target_link_libraries(network_tests PRIVATE golf_ui Qt6::Test)
  add_test(NAME online_multiplayer COMMAND network_tests)
  set_tests_properties(online_multiplayer PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
include(GNUInstallDirs)
install(TARGETS turnover-golf RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES assets/io.voncloft.TurnoverGolf.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/turnover-golf.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
